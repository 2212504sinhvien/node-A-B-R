def ghi_nhan_seq(src, seq):
    if seq is None or seq == "" or src in (None, ""):
        return 0
    try:
        s = int(seq)
    except (TypeError, ValueError):
        return 0
 
    rec = state.seq_nguon.get(src)
    if rec is None:
 
        state.seq_nguon[src] = {"min": s, "max": s, "seen": {s}}
        _tinh_lai_tong()
        return 0
 
    if rec["max"] >= 8 and s < rec["max"] // 4:
        state.seq_nguon[src] = {"min": s, "max": s, "seen": {s}}
        _tinh_lai_tong()
        return 0
 
    if s in rec["seen"]:
        _tinh_lai_tong()
        return 0
 
    rec["seen"].add(s)
    if s < rec["min"]:
        rec["min"] = s
    if s > rec["max"]:
        rec["max"] = s
 
    return _tinh_lai_tong()
 
def _tinh_lai_tong():
    truoc = state.n_lost
    du_kien = 0
    da_nhan = 0
    for r in state.seq_nguon.values():
        du_kien += (r["max"] - r["min"] + 1)
        da_nhan += len(r["seen"])
    state.n_expected = du_kien
    state.n_lost = max(0, du_kien - da_nhan)
    return max(0, state.n_lost - truoc)
 
def tinh_pdr():
    if state.n_expected <= 0:
        return None
    return (state.n_expected - state.n_lost) / state.n_expected * 100.0
 
def tach_du_lieu_do():
    from collections import defaultdict
    theo_moc = defaultdict(lambda: {"rssi": [], "snr": [],
                                    "n_msg": 0, "tx": 0, "ok": 0,
                                    "fail": 0, "hop1": 0})
    theo_rssi = []
    for r in nhat_ky.rows:
 
        if r.get("bo_qua") not in ("", None):
            continue
        kc = r.get("khoang_cach")
        if kc in ("", None):
            continue
        try:
            k = int(float(kc))
        except (TypeError, ValueError):
            continue
        t = r["loai"]
        d = theo_moc[k]
        if t == "msg":
            d["n_msg"] += 1
            if r["rssi"] not in ("", None):
                try:
                    rr = float(r["rssi"])
                    d["rssi"].append(rr)
                    theo_rssi.append(rr)
                except ValueError:
                    pass
            if r["snr"] not in ("", None):
                try:
                    d["snr"].append(float(r["snr"]))
                except ValueError:
                    pass
            try:
                if int(float(r.get("hop") or 0)) >= 1:
                    d["hop1"] += 1
            except (TypeError, ValueError):
                pass
        elif t == "tx":
            d["tx"] += 1
        elif t == "tx_ok":
            d["ok"] += 1
        elif t == "tx_fail":
            d["fail"] += 1
    return dict(theo_moc), theo_rssi
 
def hoi_quy_path_loss():
    import math
    theo_moc, _ = tach_du_lieu_do()
 
    diem = []
    for k in sorted(theo_moc):
        v = theo_moc[k]["rssi"]
        if v:
            diem.append((k, sum(v) / len(v), len(v)))
    if len(diem) < 2:
        return {"ok": False,
                "loi": "Cần ít nhất 2 mốc khoảng cách có dữ liệu RSSI "
                       "(hiện có %d). Hãy đo và chốt mốc trước." % len(diem)}
 
    xs = [math.log10(d) for d, _, _ in diem]
    ys = [r for _, r, _ in diem]
    nn = len(xs)
    sx, sy = sum(xs), sum(ys)
    sxx = sum(x * x for x in xs)
    sxy = sum(x * y for x, y in zip(xs, ys))
    den = nn * sxx - sx * sx
    if abs(den) < 1e-9:
        return {"ok": False, "loi": "Các mốc khoảng cách quá gần nhau, "
                                    "không hồi quy được."}
    a = (nn * sxy - sx * sy) / den
    b = (sy - a * sx) / nn
    n = -a / 10.0
 
    resid = [y - (a * x + b) for x, y in zip(xs, ys)]
    mres = sum(resid) / nn
    sigma = math.sqrt(sum((r - mres) ** 2 for r in resid) / nn) if nn > 1 else 0.0
 
    ytb = sy / nn
    ss_tot = sum((y - ytb) ** 2 for y in ys)
    ss_res = sum((y - (a * x + b)) ** 2 for x, y in zip(xs, ys))
    r2 = 1 - ss_res / ss_tot if ss_tot > 1e-9 else 1.0
 
    moc_kep = []
    for i in range(1, len(diem)):
        d_truoc, r_truoc, _ = diem[i - 1]
        d_sau, r_sau, _ = diem[i]
        if r_sau > r_truoc + 0.5:
            moc_kep.append(d_sau)
 
    return {"ok": True, "A": round(b, 2), "n": round(n, 2),
            "sigma": round(sigma, 2), "r2": round(r2, 3),
            "diem": diem, "moc_kep": moc_kep}
 
def thong_ke_pdr_theo_rssi(bin_db=3):
 
    from collections import defaultdict
    _, theo_rssi = tach_du_lieu_do()
    if not theo_rssi:
        return []
    lo = int(min(theo_rssi))
    hi = int(max(theo_rssi))
 
    lo = (lo // bin_db) * bin_db
    bins = defaultdict(int)
    for r in theo_rssi:
        b = (int(r) // bin_db) * bin_db
        bins[b] += 1
    out = []
    for b in sorted(bins):
        out.append((b, b + bin_db, bins[b]))
    return out

def xep_loai_link(rssi):
    if rssi is None or rssi == "":
        return ""
    try:
        r = float(rssi)
    except (TypeError, ValueError):
        return ""
    if r >= -90:
        return "Tốt"
    if r >= -110:
        return "Trung bình"
    if r >= -125:
        return "Yếu"
    return "Rất yếu"
 
def ghi_chu_su_kien(ev):
    t = ev.get("type", "")
    if t == "tx_ok":
        return "Đã nhận ACK từ nút đích"
    if t == "tx_fail":
        return "Không nhận được ACK sau khi thử lại"
    if t == "alarm":
        return "Nút %s đang phát báo động khẩn cấp" % ev.get("src", "?")
    if t == "alarm_tx":
        return "Trạm cổng tự phát báo động (lần %s)" % ev.get("attempt", "?")
    if t == "clear":
        return "Nút %s đã hủy báo động" % ev.get("src", "?")
    if t == "msg":
        hop = ev.get("hop", "")
        if hop in (0, "0"):
            return "Liên lạc trực tiếp"
        if hop not in ("", None):
            return "Qua %s trạm chuyển tiếp" % hop
    if t == "marker":
        return "Mốc %s m — bắt đầu đo" % (ev.get("text") or "?")
    return ""
 
def _fmt_num(v, nd=1):
    if v is None or v == "":
        return ""
    try:
        return round(float(v), nd)
    except (TypeError, ValueError):
        return ""
 
def _fmt(v, nd=1):
    if v is None or v == "":
        return ""
    try:
        return ("%%.%df" % nd) % float(v)
    except (TypeError, ValueError):
        return str(v)

class CsvWriter:
 
    def __init__(self, path):
        self.path = path
        self._locked_warned = False
        self._chuan_bi_file()
 
    def _chuan_bi_file(self):
        import os
        can_tao = True
        if os.path.isfile(self.path):
            try:
 
                with open(self.path, "r", encoding="utf-8-sig") as f:
                    hdr = f.readline().strip()
                if hdr and hdr.split(",")[0].strip().strip('"') == CSV_HEADER[0]:
                    can_tao = False
                else:
 
                    old = self.path.replace(
                        ".csv",
                        "_cu_" + datetime.datetime.now().strftime("%Y%m%d_%H%M%S")
                        + ".csv")
                    try:
                        os.rename(self.path, old)
                        print("  [i] log.csv dinh dang cu -> luu tai %s"
                              % os.path.basename(old))
                    except OSError:
                        pass
            except (PermissionError, OSError):
                print("  [!] log.csv dang bi mo trong Excel - dong lai de ghi.")
        if can_tao:
            try:
                with open(self.path, "w", encoding="utf-8-sig", newline="") as f:
                    csv.writer(f).writerow(CSV_HEADER)
            except PermissionError:
                print("  [!] log.csv dang bi mo trong Excel - dong lai de ghi.")
 
    def write(self, rec):
        row = [rec["stt"], rec["ngay"], rec["gio"], rec["khoang_cach"],
               rec["sf"], rec["su_kien"], rec["chieu"], rec["nguon"],
               rec["noi_dung"], rec["rssi"], rec["snr"], rec["chat_luong"],
               rec["hop"], rec["seq"], rec["rtt"], rec["ghi_chu"],
               rec.get("bo_qua", "")]
        try:
            with open(self.path, "a", encoding="utf-8-sig", newline="") as f:
                csv.writer(f).writerow(row)
            self._locked_warned = False
        except PermissionError:
            if not self._locked_warned:
                print("  [!] log.csv bi khoa (Excel dang mo?). Dashboard van chay.")
                self._locked_warned = True
 
class XlsxWriter:
 
    MIN_INTERVAL_S = 1.5
 
    def __init__(self, path, nhat_ky):
        self.path = path
        self.nhat_ky = nhat_ky
        self._last = 0.0
        self._locked_warned = False
 
        try:
            import openpyxl
            self._co_openpyxl = True
        except ImportError:
            self._co_openpyxl = False
            print("  [!] Chua cai openpyxl -> khong xuat duoc log.xlsx.")
            print("      Cai bang:  py -m pip install openpyxl")
 
    def write(self, force=False):
        if not self._co_openpyxl:
            return
        now = time.time()
        if not force and (now - self._last) < self.MIN_INTERVAL_S:
            return
        self._last = now
        try:
            self._build()
            self._locked_warned = False
        except PermissionError:
            if not self._locked_warned:
                print("  [!] log.xlsx bi khoa (Excel dang mo?). Dong lai de cap nhat.")
                self._locked_warned = True
        except Exception as e:
            if not self._locked_warned:
                print("  [!] Khong ghi duoc log.xlsx: %s" % e)
                self._locked_warned = True

class LogParser:
    def __init__(self, emit_fn):
        self.emit = emit_fn
        self.pending_rx = None
        self.pending_alarm = None
 
    def feed(self, line):
        line = line.rstrip("\r\n")
        ts = datetime.datetime.now().strftime("%H:%M:%S")
 
        m = CFG_RE.search(line)
        if m:
            with state.lock:
                state.cfg_sf   = int(m.group(2))
                state.cfg_bw   = float(m.group(3))
                state.cfg_cr   = m.group(4)
                state.cfg_freq = float(m.group(5))
                state.cfg_pwr  = int(m.group(6))
                state.cfg_rep  = int(m.group(7))
                state.cfg_node = m.group(1)
            print("  [CFG] Node '%s' | SF=%s | BW=%s kHz | CR=4/%s | %s MHz"
                  % (m.group(1), m.group(2), m.group(3), m.group(4), m.group(5)))
            return
 
        m = RX_RE.search(line)
        if m:
            self.pending_rx = {
                "type": "msg", "num": int(m.group(1)), "text": m.group(2),
            }
            return
 
        m = SRC_RE.search(line)
        if m and self.pending_rx:
            self.pending_rx["src"] = m.group(1)
            self.pending_rx["hop"] = int(m.group(2))
            self.pending_rx["seq"] = int(m.group(3))
            return
 
        m = RSSI_RE.search(line)
        if m:
            rssi, snr = float(m.group(1)), float(m.group(2))
            if self.pending_rx:
                self.pending_rx["rssi"] = rssi
                self.pending_rx["snr"] = snr
                self._dispatch(self.pending_rx, ts)
                self.pending_rx = None
            elif self.pending_alarm:
                self.pending_alarm["rssi"] = rssi
                self.pending_alarm["snr"] = snr
                self._dispatch(self.pending_alarm, ts)
                self.pending_alarm = None
            return
 
        m = ALARM_RE.search(line)
        if m:
            ev = {"type": "alarm", "src": m.group(1)}
            mr = ALARM_RSSI_RE.search(line)
            if mr:
                ev["rssi"] = float(mr.group(1))
                ev["snr"]  = float(mr.group(2))
                self._dispatch(ev, ts)
            else:
 
                self.pending_alarm = ev
            return
 
        m = ALARM_RSSI_RE.search(line)
        if m and self.pending_alarm:
            self.pending_alarm["rssi"] = float(m.group(1))
            self.pending_alarm["snr"]  = float(m.group(2))
            self._dispatch(self.pending_alarm, ts)
            self.pending_alarm = None
            return
 
        m = ALARMTXCLEAR_RE.search(line)
        if m:
            self._dispatch({"type": "clear", "src": m.group(1)}, ts)
            return
 
        m = CLEAR_RE.search(line)
        if m:
            self._dispatch({"type": "clear", "src": m.group(1)}, ts)
            return
 
        m = TX_RE.search(line)
        if m:
            self._dispatch({
                "type": "tx", "num": int(m.group(1)), "text": m.group(2),
            }, ts)
            return
 
        m = TXOK_RE.search(line)
        if m:
            self._dispatch({"type": "tx_ok", "rtt_ms": int(m.group(1))}, ts)
            return
 
        m = TXFAIL_RE.search(line)
        if m:
            self._dispatch({"type": "tx_fail"}, ts)
            return
 
        m = ALARMTX_RE.search(line)
        if m:
            self._dispatch({
                "type": "alarm_tx",
                "attempt": int(m.group(1)),
                "seq": int(m.group(2)),
            }, ts)
            return
 
    def _dispatch(self, ev, ts):
        ev["ts"] = ts
 
        with state.lock:
            t = ev["type"]
            if t == "msg":
                state.n_rx += 1
                state.last_rssi = ev.get("rssi")
                state.last_snr = ev.get("snr")
            elif t == "alarm":
                state.n_alarm += 1
                state.alarm_active = True
                state.alarm_src = ev.get("src")
                state.alarm_ts = datetime.datetime.now()
                state.last_rssi = ev.get("rssi")
                state.last_snr = ev.get("snr")
            elif t == "clear":
                state.n_clear += 1
                state.alarm_active = False
                state.alarm_src = None
            elif t == "tx":
                pass
            elif t == "tx_ok":
                state.n_tx_ok += 1
                state.last_rtt = ev.get("rtt_ms")
            elif t == "tx_fail":
                state.n_tx_fail += 1
            elif t == "alarm_tx":
 
                state.alarm_active = True
                state.alarm_src    = "B"
                state.alarm_ts     = datetime.datetime.now()
 
                state.n_alarm_tx = getattr(state, "n_alarm_tx", 0) + 1
 
            if t == "msg":
                ghi_nhan_seq(ev.get("src"), ev.get("seq"))
 
            log_entry = {
                "ts": ts,
                "type": t,
                "src": ev.get("src", ""),
                "text": ev.get("text", ""),
                "rssi": ev.get("rssi", ""),
                "snr": ev.get("snr", ""),
                "hop": ev.get("hop", ""),
                "seq": ev.get("seq", ""),
                "rtt_ms": ev.get("rtt_ms", ""),
            }
            state.logs.appendleft(log_entry)
 
            if t in ("msg", "alarm") and ev.get("rssi") is not None:
                state.chart_data.append({
                    "ts": ts,
                    "rssi": ev["rssi"],
                    "snr": ev.get("snr", ""),
                    "type": t,
                })
 
        rec = nhat_ky.them(ev)
        csv_writer.write(rec)
        xlsx_writer.write()
 
        with state.lock:
            stats_snapshot = {
                "n_rx": state.n_rx,
                "n_tx_ok": state.n_tx_ok,
                "n_tx_fail": state.n_tx_fail,
                "n_alarm": state.n_alarm,
                "n_alarm_tx": state.n_alarm_tx,
                "n_clear": state.n_clear,
                "last_rssi": state.last_rssi,
                "last_snr": state.last_snr,
                "last_rtt": state.last_rtt,
                "alarm_active": state.alarm_active,
                "alarm_src": state.alarm_src,
                "n_lost": state.n_lost,
                "n_expected": state.n_expected,

// --- WEB AUDIO ALARM ---
let audioCtx = null;
let alarmOscillator = null;
 
function startAlarmSound() {
  if (alarmOscillator) return;
  try {
    audioCtx = audioCtx || new (window.AudioContext || window.webkitAudioContext)();
    alarmOscillator = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    alarmOscillator.type = 'square';
    alarmOscillator.frequency.value = 880;
    gain.gain.value = 0.15;
    alarmOscillator.connect(gain);
    gain.connect(audioCtx.destination);
    alarmOscillator.start();
    // Modulate: on/off pattern
    const lfo = audioCtx.createOscillator();
    lfo.frequency.value = 3;
    const lfoGain = audioCtx.createGain();
    lfoGain.gain.value = 0.15;
    lfo.connect(lfoGain);
    lfoGain.connect(gain.gain);
    lfo.start();
    alarmOscillator._lfo = lfo;
  } catch(e) { console.log('Audio not available:', e); }
}
 
function stopAlarmSound() {
  if (alarmOscillator) {
    try { alarmOscillator.stop(); alarmOscillator._lfo && alarmOscillator._lfo.stop(); } catch(e){}
    alarmOscillator = null;
  }
}

function updateStats(s) {
  document.getElementById('nRx').textContent = s.n_rx;
  document.getElementById('nTxOk').textContent = s.n_tx_ok;
  document.getElementById('nTxFail').textContent = s.n_tx_fail;

  document.getElementById('nClear').textContent = s.n_clear;
  document.getElementById('lastRssi').textContent = s.last_rssi != null ? s.last_rssi + ' dBm' : '-';
  document.getElementById('lastSnr').textContent = s.last_snr != null ? 'SNR: ' + s.last_snr + ' dB' : 'SNR: -';
  document.getElementById('lastRtt').textContent = s.last_rtt != null ? 'RTT đầy đủ: ' + s.last_rtt + ' ms' : 'RTT đầy đủ: -';
 
  // --- TO MAU THE RSSI/SNR THEO NGUONG THUC NGHIEM ---
  // Nguong do tu thuc nghiem: > -120 tot, -127..-120 vung xam, < -127 mat.
  const rssiCard = document.getElementById('rssiCard');
  rssiCard.classList.remove('good', 'mid', 'bad');
  if (s.last_rssi != null) {
    const r = parseFloat(s.last_rssi);
    if (r > -120) rssiCard.classList.add('good');
    else if (r >= -127) rssiCard.classList.add('mid');
    else rssiCard.classList.add('bad');
  }
 
  // --- CAU HINH RF (SF, BW, tan so) ---
  const cfgSfEl = document.getElementById('cfgSf');
  const cfgInfoEl = document.getElementById('cfgInfo');
  if (s.sf != null) {
    cfgSfEl.textContent = 'SF' + s.sf;
    cfgInfoEl.textContent = (s.bw || '?') + ' kHz | '
      + (s.freq || '?') + ' MHz | ' + (s.rep ? 'co repeater' : 'khong rep');
    // SF thap = tam xa ngan -> to mau canh bao
    cfgSfEl.style.color = (s.sf >= 11) ? '#22c55e'

  } else {
    cfgSfEl.textContent = '-';
    cfgInfoEl.textContent = 'Chua doc duoc [CFG]';
  }
  const pdrEl = document.getElementById('pdrVal');
  const lostEl = document.getElementById('lostVal');
  const pdrCard = document.getElementById('pdrCard');
  pdrCard.classList.remove('good', 'mid', 'bad');

    pdrEl.textContent = '-';
    lostEl.textContent = 'Chua du du lieu';
  } else {
    pdrEl.textContent = s.pdr.toFixed(1) + '%';

  const dot = document.getElementById('statusDot');

  const alarmCard = document.getElementById('alarmCard');

function addLogEntry(ev) {
  const list = document.getElementById('logList');
  const div = document.createElement('div');
  div.className = 'log-entry';
 
  const typeLabels = {
    msg: 'Node A', tx: 'Node B', tx_ok: 'B OK', tx_fail: 'B FAIL',
    alarm: 'BAO CHAY', clear: 'HUY', alarm_tx: 'B SOS', marker: 'MOC DO'
  };
  const typeClass = ev.type || 'msg';
  const label = typeLabels[typeClass] || ev.type;
 
  let body = '';
  if (ev.type === 'msg') body = '"' + (ev.text||'') + '" tu Node ' + (ev.src||'?');
  else if (ev.type === 'tx') body = 'Node B gui: "' + (ev.text||'') + '"';
  else if (ev.type === 'tx_ok') body = 'Node B gui thanh cong | RTT đầy đủ ' + (ev.rtt_ms||'?') + ' ms (đi+về)';
  else if (ev.type === 'tx_fail') body = 'Node B gui that bai - khong co ACK';
  else if (ev.type === 'alarm') body = '!!! BAO CHAY tu Node ' + (ev.src||'?') + ' !!!';
  else if (ev.type === 'alarm_tx') body = '!!! GATEWAY (Node B) TU PHAT BAO DONG - lan ' + (ev.attempt||'?') + ' !!!';
  else if (ev.type === 'clear') body = (ev.src === 'watchdog' ? '[Timeout 45s] Tu tat canh bao' : 'Node ' + (ev.src||'?') + ' da huy bao dong');
  else if (ev.type === 'marker') body = '=== CHOT MOC ' + (ev.text||'?') + ' m ===';
 
  let rssiStr = '';
  if (ev.rssi !== '' && ev.rssi != null) rssiStr = ev.rssi + ' dBm';
 
  div.innerHTML = '<span class="log-ts">' + (ev.ts||'') + '</span>'
    + '<span class="log-type ' + typeClass + '">' + label + '</span>'
    + '<span class="log-body">' + body + '</span>'
    + '<span class="log-rssi">' + rssiStr + '</span>';
 
  list.insertBefore(div, list.firstChild);
 
  // Gioi han so dong hien thi
  while (list.children.length > 200) list.removeChild(list.lastChild);
}
 
function addChartPoint(ev) {

  chart.data.datasets[1].data.push(snr);
  // Gioi han so diem
  if (chart.data.labels.length > 100) {
    chart.data.labels.shift();
    chart.data.datasets[0].data.shift();
    chart.data.datasets[1].data.shift();
  }
  chart.update('none');
}

def serial_reader(port):
    import serial
    try:
        ser = serial.Serial(port, BAUDRATE, timeout=1)
        print(f"  Da mo {port} @ {BAUDRATE} baud")
    except serial.SerialException as e:
        print(f"  [!] Khong mo duoc {port}: {e}")
        return
 
    while True:
        try:
            line = ser.readline().decode("utf-8", errors="replace")
            if line:
                parser.feed(line)
        except Exception as e:
            print(f"  [!] Loi doc Serial: {e}")
            time.sleep(1)
 
def find_port():
    try:
        import serial.tools.list_ports
    except ImportError:
        print("  [!] Chua cai pyserial: pip install pyserial")
        return None
 
    ports = serial.tools.list_ports.comports()
    if not ports:
        return None
 
    for p in ports:
        desc = (p.description or "").lower()
        if "nrf" in desc or "adafruit" in desc:
            return p.device
 
    generic = [p for p in ports
               if "usb serial device" in (p.description or "").lower()]
    if len(generic) == 1 and len(ports) == 1:
        print(f"  [i] Nhan dang '{generic[0].description}' (nRF SuperMini)")
        return generic[0].device
 
    print("  Cac cong COM hien co:")
    for p in ports:
        print(f"    {p.device}  -  {p.description}")
    return None

def alarm_watchdog():
    ALARM_HOLD = 45.0
    import time as _t
    while True:
        _t.sleep(1.0)
        with state.lock:
            if state.alarm_active and state.alarm_ts is not None:
                age = (datetime.datetime.now() - state.alarm_ts).total_seconds()
                if age > ALARM_HOLD:
                    state.alarm_active = False
                    state.alarm_src = None
                    print("  [WATCHDOG] Het 45s khong nghe goi alarm -> tu tat canh bao web")
 
                    socketio.emit("event", {
                        "log": {
                            "ts": datetime.datetime.now().strftime("%H:%M:%S"),
                            "type": "clear",
                            "src": "watchdog",
                            "text": "",
                            "rssi": "", "snr": "",
                            "hop": "", "seq": "", "rtt_ms": "",
                        },
                        "stats": {
                            "n_rx": state.n_rx,
                            "n_tx_ok": state.n_tx_ok,
                            "n_tx_fail": state.n_tx_fail,
                            "n_alarm": state.n_alarm,
                            "n_alarm_tx": state.n_alarm_tx,
                            "n_clear": state.n_clear,
                            "last_rssi": state.last_rssi,
                            "last_snr": state.last_snr,
                            "last_rtt": state.last_rtt,
                            "alarm_active": False,
                            "alarm_src": None,
                        },
                    })
